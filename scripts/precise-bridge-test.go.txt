package main

import (
	"encoding/json"
	"strings"
	"testing"
)

func TestVPN013PreciseBridgeJSON(t *testing.T) {
	key := strings.Repeat("12", 32)
	uapi := "private_key=" + key + "\nr013_legacy15=false\npublic_key=" + key + "\nendpoint=127.0.0.1:8443\nallowed_ip=0.0.0.0/0\n"
	request, _ := json.Marshal(map[string]interface{}{"uapi": uapi, "addresses": []string{"10.0.0.2/32"}, "mtu": 1280})
	converted := r013ConvertWireGuardJSON(request)
	var out struct {
		Secret string `json:"secretKey"`
		Peers  []struct {
			Endpoint string `json:"endpoint"`
		} `json:"peers"`
	}
	if json.Unmarshal(converted, &out) != nil || out.Secret != key || len(out.Peers) != 1 || out.Peers[0].Endpoint != "127.0.0.1:8443" {
		t.Fatal("valid synthetic configuration not preserved")
	}
	for _, bad := range [][]byte{
		[]byte(`{"uapi":"bad-secret-marker","addresses":["10.0.0.2/32"],"mtu":1280}`),
		[]byte(`{"uapi":"bad-secret-marker","addresses":["10.0.0.2/32"],"mtu":1280,"extra":1}`),
		append(append([]byte(nil), request...), []byte(` {}`)...),
	} {
		if result := r013ConvertWireGuardJSON(bad); result != nil {
			t.Fatal("invalid data accepted")
		}
	}
}
