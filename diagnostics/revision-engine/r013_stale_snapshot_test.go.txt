func TestVPN013LegacyStaleSnapshotAfterModeSwitch(t *testing.T) {
	pair := genTestPair(t, false)
	d := pair[0].dev
	if err := d.IpcSet(uapiCfg("r013_legacy15", "true", "i1", "<b 0x01>", "itime", "60")); err != nil {
		t.Fatal(err)
	}
	d.obfuscation.RLock()
	old := d.legacyConfig
	d.obfuscation.RUnlock()
	if err := d.IpcSet(uapiCfg("r013_legacy15", "false", "i1", "<b 0x02>")); err != nil {
		t.Fatal(err)
	}
	bufs, err := d.r013BuildLegacyInitial(old, 0, 0, 0)
	if err == nil && len(bufs) > 0 {
		t.Fatal("old-mode packets prepared after modern mode was committed")
	}
}

func TestVPN013LegacyRejectPreparedPacketAfterModeSwitch(t *testing.T) {
	pair := genTestPair(t, false)
	d := pair[0].dev
	if err := d.IpcSet(uapiCfg("r013_legacy15", "true", "i1", "<b 0x01>", "itime", "60")); err != nil {
		t.Fatal(err)
	}
	bufs, _, legacy, revision, err := d.r013PrepareInitialObfuscationMode()
	if err != nil || !legacy || len(bufs) == 0 {
		t.Fatal("legacy preparation failed")
	}
	if err := d.IpcSet(uapiCfg("r013_legacy15", "false", "i1", "<b 0x02>")); err != nil {
		t.Fatal(err)
	}
	var peer *Peer
	for _, p := range d.peers.keyMap {
		peer = p
		break
	}
	if peer == nil {
		t.Fatal("test peer missing")
	}
	if err := peer.r013SendLegacyJunk(bufs, revision); err != errR013LegacyStale {
		t.Fatal("old-mode packet was not rejected before network send")
	}
}

type r013BlockingBind struct {
	conn.Bind
	entered chan struct{}
	release chan struct{}
}

func (b *r013BlockingBind) Send(_ [][]byte, _ conn.Endpoint) error {
	close(b.entered)
	<-b.release
	return nil
}

func TestVPN013LegacySendCompletesBeforeModeCommit(t *testing.T) {
	configs, endpoints := genConfigs(t)
	binds := bindtest.NewChannelBinds()
	blocker := &r013BlockingBind{Bind: binds[0], entered: make(chan struct{}), release: make(chan struct{})}
	tun := tuntest.NewChannelTUN()
	d := NewDevice(tun.TUN(), blocker, NewLogger(LogLevelError, "synthetic: "))
	t.Cleanup(d.Close)
	if err := d.IpcSet(configs[0]); err != nil {
		t.Fatal(err)
	}
	if err := d.Up(); err != nil {
		t.Fatal(err)
	}
	if err := d.IpcSet(fmt.Sprintf(endpoints[0], 12345)); err != nil {
		t.Fatal(err)
	}
	if err := d.IpcSet(uapiCfg("r013_legacy15", "true", "i1", "<b 0x01>", "itime", "60")); err != nil {
		t.Fatal(err)
	}
	bufs, _, legacy, revision, err := d.r013PrepareInitialObfuscationMode()
	if err != nil || !legacy || len(bufs) == 0 {
		t.Fatal("legacy preparation failed")
	}
	var peer *Peer
	for _, p := range d.peers.keyMap {
		peer = p
		break
	}
	if peer == nil {
		t.Fatal("test peer missing")
	}
	release := sync.OnceFunc(func() { close(blocker.release) })
	defer release()
	sendResult := make(chan error, 1)
	go func() { sendResult <- peer.r013SendLegacyJunk(bufs, revision) }()
	select {
	case <-blocker.entered:
	case <-time.After(time.Second):
		t.Fatal("send did not enter test bind")
	}
	updateResult := make(chan error, 1)
	go func() { updateResult <- d.IpcSet(uapiCfg("r013_legacy15", "false", "i1", "<b 0x02>")) }()
	select {
	case <-updateResult:
		release()
		t.Fatal("mode committed while old packet was being sent")
	case <-time.After(50 * time.Millisecond):
	}
	release()
	select {
	case err := <-sendResult:
		if err != nil {
			t.Fatal(err)
		}
	case <-time.After(time.Second):
		t.Fatal("send did not complete")
	}
	select {
	case err := <-updateResult:
		if err != nil {
			t.Fatal(err)
		}
	case <-time.After(time.Second):
		t.Fatal("mode update did not complete")
	}
	if err := peer.r013SendLegacyJunk(bufs, revision); err != errR013LegacyStale {
		t.Fatal("old revision accepted after update")
	}
}
